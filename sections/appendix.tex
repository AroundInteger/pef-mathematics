\section{Mathematical Appendix}
\label{sec:appendix}

\subsection{Canonical Form}
\label{sec:appendix_canonical}

Start from $\eta = (1+\kappa)/(1+\kappa-2\sqrt{\kappa}\,\rho)$ and divide the numerator and the denominator by $\sqrt{\kappa}$:
\[
  \eta
  = \frac{\kappa^{-1/2}+\kappa^{1/2}}{\kappa^{-1/2}+\kappa^{1/2}-2\rho}.
\]
With $\tau = \tfrac{1}{2}\log\kappa$ one has $\kappa^{1/2} = e^{\tau}$ and $\kappa^{-1/2}+\kappa^{1/2} = 2\cosh\tau$, so
\[
  \eta = \frac{2\cosh\tau}{2\cosh\tau-2\rho} = \frac{\cosh\tau}{\cosh\tau-\rho}.
\]

\subsection{Involution}

The substitution $\kappa\mapsto 1/\kappa$ sends $\tau$ to $-\tau$.
Since $\cosh$ is even, \cref{eq:canonical} is unchanged.

\subsection{Cumulants of the Latent Family}

Let $u = e^{\theta}\in(0,1)$ and $\log\eta = -\log(1-u)$.
Then
\[
  \frac{\mathrm{d}\log\eta}{\mathrm{d}\theta}
  = \frac{u}{1-u}.
\]
From $\eta = 1/(1-u)$ one has $u = 1-1/\eta$, so $u/(1-u) = \eta-1$.
Differentiating again, $\mathrm{d}\eta/\mathrm{d}\theta = \eta^{2} u = \eta(\eta-1)$.
These are $\mathbb{E}[N]$ and $\Var(N)$ for \cref{eq:geo_family}.
They are not moments of $(X_{\mathrm{A}},X_{\mathrm{B}})$.

\subsection{Cosine Form of \texorpdfstring{$u$}{u}}

\[
  \frac{2\sqrt{\kappa}\,\rho}{1+\kappa}
  = \frac{2\rho}{e^{\tau}+e^{-\tau}}
  = \rho\,\sech\tau.
\]
Hence $\cos\sigma = u$ and $\eta = 1/(1-\cos\sigma)$ whenever that cosine is defined.
The longitude is free.

\subsection{Latent-Family Delta Method}

Let $\mathcal{I}(\theta) = \eta(\eta-1)$ and $\psi(\theta) = 2\arctanh(e^{\theta/2})$ for $\theta<0$.
Then $\mathrm{d}\psi/\mathrm{d}\theta = \sqrt{\mathcal{I}(\theta)}$.
If $\hat\theta$ is the latent-family maximum-likelihood estimator,
\[
  \Var(\hat\psi)
  \approx \Bigl(\frac{\mathrm{d}\psi}{\mathrm{d}\theta}\Bigr)^{2}\frac{1}{n\,\mathcal{I}(\theta)}
  = \frac{1}{n}.
\]
Since $\mathrm{d}\eta/\mathrm{d}\theta = \eta(\eta-1)$,
\[
  \Var(\hat\eta) \approx \frac{\eta(\eta-1)}{n}.
\]

\subsection{Plug-in Delta Method}
\label{sec:appendix_plugin}

Standard bivariate-normal expansions give $\Var(\hat\rho)\approx(1-\rho^{2})^{2}/n$ and $\Var(\log s^{2})\approx 2/n$, with $\Cov(\log s_{\mathrm{A}}^{2},\log s_{\mathrm{B}}^{2})\approx 2\rho^{2}/n$.
Hence
\[
  \Var(\hat\tau)
  = \Var\Bigl(\tfrac{1}{2}\log(s_{\mathrm{B}}^{2}/s_{\mathrm{A}}^{2})\Bigr)
  \approx \frac{1-\rho^{2}}{n}.
\]
The pair $(\hat\rho,\hat\tau-\tau)$ is a function of the standardised sample.
The swap $\mathrm{A}\leftrightarrow\mathrm{B}$ fixes $\hat\rho$ and sends $\hat\tau-\tau$ to its negative, so the asymptotic covariance is zero: $\Cov(\hat\rho,\hat\tau) = o(1/n)$.

Now $u = \rho\,\sech\tau$, $\partial u/\partial\rho = \sech\tau$, and $\partial u/\partial\tau = -u\tanh\tau$.
Therefore
\begin{align*}
  \Var(\hat u)
  &\approx \frac{\sech^{2}\tau\,(1-\rho^{2})^{2} + u^{2}\tanh^{2}\tau\,(1-\rho^{2})}{n}\\
  &= \frac{\sech^{2}\tau\,(1-\rho^{2})\,(1-u^{2})}{n}.
\end{align*}
The identity $\sech^{2}\tau\,(1-\rho^{2}) = \sech^{2}\tau-u^{2}$ is \cref{eq:var_u_plugin}.
For $\eta = 1/(1-u)$ one has $\mathrm{d}\eta/\mathrm{d}u = \eta^{2}$, which is \cref{eq:var_eta_plugin}.
For $\psi = 2\arctanh\sqrt{u}$ one has $\mathrm{d}\psi/\mathrm{d}u = 1/(\sqrt{u}\,(1-u))$, which is \cref{eq:var_psi_plugin}.
Both plug-in formulae require $u\in(0,1)$ when they invoke $\psi$ or the latent information.
\Cref{eq:var_u_plugin} itself uses only $|u|<1$.
